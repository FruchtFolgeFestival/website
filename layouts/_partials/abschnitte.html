` ($text | markdownify | plainify) }} {{- $abschnitte = slice (printf "%s%s" $vorschau (index $abschnitte 0)) | append (after 1 $abschnitte) }} {{- end }} {{- end }} {{- $vorher := "hero" }} {{- range $i, $html := $abschnitte }} {{- $bg := cond (modBool $i 2) "bg" "alt" }} {{- partial "trenner.html" (dict "von" $vorher "nach" $bg) }}
